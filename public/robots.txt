# NOTE ON STRUCTURE
# A crawler obeys exactly one group — the most specific one that matches its
# token — and ignores every other group. Listing a bot with only "Allow: /"
# therefore EXEMPTS it from the shared Disallow rules below. So named groups
# exist here only where they carry a rule that differs from the default.
# Everything else is covered by "User-agent: *", which already allows crawling.
# ============================================================================

User-agent: *
Allow: /

# Admin surfaces and write endpoints — never useful in an index.
Disallow: /admin/
Disallow: /api/
Disallow: /login

# Crawl traps: ~5,850 thin near-duplicates over ?profile= x ?type=.
Disallow: /suggest-edit
Disallow: /search

# Legacy URLs from the previous site on this domain. Gone, not moved.
Disallow: /services/
Disallow: /industries/

# ============================================================================
# THROTTLED CRAWLERS

# the default one.
